// ---------------------------------------------------------------------------
// Capacities (fixed backing storage; growth is a one-line change per cap).
// Overflow anywhere is fail-closed via MliFunction.overflowed — never a
// silent drop past a wall (machine_ir MIR_MAX_INSTRS lesson).
// ---------------------------------------------------------------------------

pub let MLI_MAX_BLOCKS: i64 = 16

// Instruction pool geometry (see "storage layout" note below): one stride per
// stored as a struct-of-arrays pool on MliFunction (the same shape main's IR
// arena adopted), one column per instruction field, and MliInst exists as an
// in-memory VALUE type assembled/disassembled by mli_inst_get / mli_inst_put.
// Block b owns pool slots [b*MLI_BLOCK_STRIDE, b*MLI_BLOCK_STRIDE+MLI_MAX_INSTRS)
// for its body and slot b*MLI_BLOCK_STRIDE+MLI_MAX_INSTRS for its terminator.
// ---------------------------------------------------------------------------

// MliBlock is an API view assembled by mli_block_get; block metadata is
// STORED as scalar SoA columns on MliFunction (blk_id / blk_instr_count /
// blk_has_term). Measured on this lane: scalar field writes through
// &!MliFunction into an array-of-struct element (f.blocks[b].field = v) also
