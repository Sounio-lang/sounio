fn ty_slot_count(ty: i64, ty_hash: i64) -> i64 with Panic, Div, Mut {
    if ty == 8 { return arr_storage_slots(ty_hash) }
    if ty == 6 {
        // Size-hint sentinel: 1000000007 + nslots, set after .0 tuple field access
        if ty_hash >= 1000000007 && ty_hash < 1000001007 { return ty_hash - 1000000007 }
        let si = st_find(ty_hash)
        if si >= 0 { return st_total_size(si) / 8 }
    }
    if ty == 0 && ty_hash != 0 {
        let fwd_si = st_find(ty_hash)
        if fwd_si >= 0 { return st_total_size(fwd_si) / 8 }
    }
    if type_is_option_inline(ty, ty_hash) { return 2 }
    if ty == 11 { return 1 }
    if ty == 13 { return 2 }
    if knowledge_hash_is(ty_hash) { return 3 }
    return 1
}
