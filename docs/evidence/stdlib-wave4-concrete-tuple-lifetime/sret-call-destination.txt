fn emit_sret_destination_x86(ret_slots: i64) with Mut, Panic, Div {
    if ret_slots * 8 >= local_bss_spill_bytes() {
        let sret_addr = GL_BSS_BASE + bss_alloc_aligned(ret_slots * 8)
        em(0x48); em(0xb8); em64(sret_addr)
        return
    }
    let ret_base = NEXT_SLOT
    NEXT_SLOT = NEXT_SLOT + ret_slots
    em(0x48); em(0x8d); em(0x85)
    em32(0 - (ret_base + ret_slots - 1) * 8)
    return
}
