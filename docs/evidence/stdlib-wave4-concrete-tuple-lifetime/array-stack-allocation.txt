                    em32(0 - (base + total_slots - 1) * 8)
                } else {
                    em(0x31); em(0xc0)
                }
            } else {
                // Scalar element: rep stosq fills all slots with RAX
                let nslots = arr_storage_slots(arr_hash)
                if nslots > 0 {
                    let base = NEXT_SLOT
                    NEXT_SLOT = NEXT_SLOT + nslots
                    em(0x48); em(0x8d); em(0xbd)
                    em32(0 - (base + nslots - 1) * 8)
                    em(0xb9); em32(nslots)
                    em(0xf3); em(0x48); em(0xab)
                    em(0x48); em(0x8d); em(0x85)
                    em32(0 - (base + nslots - 1) * 8)
                } else {
                    em(0x31); em(0xc0)
                }
            }
            EXPR_IS_F64 = 0
            EXPR_TY = 8
            EXPR_TY_HASH = arr_hash
            LAST_ALEN = rep_count
            return
        }
