use stats::regression::nb2_likelihood::{nb2_likelihood_difference_slot,nb2_cached_likelihood_difference_slot,nb2_log_probability_mean_difference_into,nb2_log_probability_mean_rank_difference_into,nb2_log_probability_mean_rank_cached_difference_into,nb2_likelihood_difference_value,nb2_cached_likelihood_difference_value}
use collections::numeric_buffer::{numeric_buffer_new,numeric_buffer_set,numeric_buffer_close}
use collections::integer_buffer::{integer_buffer_new,integer_buffer_set,integer_buffer_close}
fn one(kind:i64) with IO,Alloc,Mut,Div,Panic {
 let n:i64=if kind==2 {2}else{3}
 var counts=integer_buffer_new(n,24)
 var first=numeric_buffer_new(n,24)
 var second=numeric_buffer_new(n,24)
 var i:i64=0
 while i<n {
  var a:f64=1.0
  var b:f64=2.0
  if kind==0 {
   if i==0 {a=1e-300;b=2e-300}
   if i==1 {a=2.0;b=1.0}
  }
  if kind==1 {
   if i==1 {a=2.0;b=1.0}
   if i==2 {a=1e-300;b=2e-300}
  }
  if kind==2 {if i==1 {a=2.0;b=1.0}}
  integer_buffer_set(&!counts,i,0)
  numeric_buffer_set(&!first,i,a)
  numeric_buffer_set(&!second,i,b)
  let ba:i64=f64_to_bits(a)
  let bb:i64=f64_to_bits(b)
  print("INPUT ");print(kind);print(" ");print(i);print(" ");print(ba);print(" ");println(bb)
  i=i+1
 }
 var s=nb2_likelihood_difference_slot()
 var c=nb2_cached_likelihood_difference_slot()
 let legacy:i64=nb2_log_probability_mean_difference_into(&counts,&first,0.0,&second,0.0,n,&!s)
 let lv:f64=nb2_likelihood_difference_value(&s)
 let lb:i64=f64_to_bits(lv)
 print("NATIVE ");print(kind);print(" legacy ");print(legacy);print(" ");println(lb)
 let rank:i64=nb2_log_probability_mean_rank_difference_into(&counts,&first,0.0,&second,0.0,n,&!s)
 let rv:f64=nb2_likelihood_difference_value(&s)
 let rb:i64=f64_to_bits(rv)
 print("NATIVE ");print(kind);print(" rank ");print(rank);print(" ");println(rb)
 let cached:i64=nb2_log_probability_mean_rank_cached_difference_into(&counts,&first,0.0,&second,0.0,n,&!c)
 let cv:f64=nb2_cached_likelihood_difference_value(&c)
 let cb:i64=f64_to_bits(cv)
 print("NATIVE ");print(kind);print(" cached ");print(cached);print(" ");println(cb)
 integer_buffer_close(counts);numeric_buffer_close(first);numeric_buffer_close(second)
}
fn main()->i32 with IO,Alloc,Mut,Div,Panic {
 one(0);one(1);one(2)
 println("AGGREGATE DIAGNOSTIC COMPLETE")
 0
}
