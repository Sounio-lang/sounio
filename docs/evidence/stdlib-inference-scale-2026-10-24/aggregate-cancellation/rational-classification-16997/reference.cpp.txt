R exact(std::int64_t signedbits){auto u=std::bit_cast<std::uint64_t>(signedbits);auto e=(u>>52)&2047; if(e==2047)throw std::runtime_error("nonfinite native input");cpp_int m=u&((std::uint64_t(1)<<52)-1);if(e)m+=cpp_int(1)<<52;int power=e?int(e)-1023-52:-1074;R v=m;if(power>=0)v*=cpp_int(1)<<power;else v/=cpp_int(1)<<(-power);return u>>63?-v:v;}
int main(int argc,char**argv){try{if(argc!=2)throw std::runtime_error("stdout required");std::ifstream f(argv[1]);if(!f)throw std::runtime_error("open");std::map<int,R>sums;std::map<int,int>counts;std::string line;bool marker=false;int outputs=0;std::cout<<std::setprecision(60);while(std::getline(f,line)){std::istringstream in(line);std::string tag;in>>tag;if(tag=="INPUT"){int k,i;std::int64_t a,b;in>>k>>i>>a>>b;if(!in||i!=counts[k])throw std::runtime_error("input order");sums[k]+=exact(a)-exact(b);counts[k]++;}else if(tag=="NATIVE"){int k,status;std::string mode;std::int64_t bits;in>>k>>mode>>status>>bits;if(!in||counts[k]!=(k==2?2:3))throw std::runtime_error("output");R truth=sums[k]/counts[k],native=exact(bits); const bool nonzero=numerator(truth)!=0; const bool zero_native=(bits==0||bits==INT64_MIN); const bool false_zero=status==0&&zero_native&&nonzero;std::cout<<"CASE\t"<<k<<'\t'<<mode<<"\tstatus\t"<<status<<"\texact_mean\t"<<truth.convert_to<D>()<<"\tnative\t"<<native.convert_to<D>()<<"\tfalse_available_zero\t"<<false_zero<<"\tnumerator_nonzero_before_conversion\t"<<nonzero<<'\n';outputs++;}else if(line=="AGGREGATE DIAGNOSTIC COMPLETE")marker=true;else if(!line.empty())throw std::runtime_error("unknown line");}if(!marker||outputs!=9||counts.size()!=3)throw std::runtime_error("incomplete");std::cout<<"EXACT BINARY RATIONAL DIAGNOSTIC COMPLETE; classification only, no acceptance claim\n";return 0;}catch(const std::exception&e){std::cerr<<e.what()<<'\n';return 2;}}
