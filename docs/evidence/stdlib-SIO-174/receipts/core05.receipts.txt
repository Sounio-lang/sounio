 var r=Dp54Times{status:2,start:start,t2:0.0,t3:0.0,t4:0.0,t5:0.0,end:end,h:h}
 if !is_finite(h)||!is_finite(start)||!is_finite(end) {r.status=4;return r}
 if h==0.0 {return r}
 r.t2=start+(1.0/5.0)*h;r.t3=start+(3.0/10.0)*h;r.t4=start+(4.0/5.0)*h;r.t5=start+(8.0/9.0)*h
 if !is_finite(r.t2)||!is_finite(r.t3)||!is_finite(r.t4)||!is_finite(r.t5) {r.status=4;return r}
 if h>0.0 {if !(start<r.t2&&r.t2<r.t3&&r.t3<r.t4&&r.t4<r.t5&&r.t5<end) {return r}}else{if !(start>r.t2&&r.t2>r.t3&&r.t3>r.t4&&r.t4>r.t5&&r.t5>end) {return r}}
 r.status=0;r
}
// No inverse bitcast: arithmetic neighbor, separately qualified against C++23
// std::nextafter on both the committed seed and current modular compiler.
fn dp_adjacent(x:f64,direction:f64)->f64 with Mut {
 let magnitude=dp_abs(x)
 var delta=magnitude*1.1102230246251565e-16
 if delta<4.9406564584124654e-324 {delta=4.9406564584124654e-324}
 var y=x+direction*delta
 if y==x {y=x+direction*(2.0*delta)}
 if y==0.0&&x<0.0 {return -0.0}
 y
}
fn dp_plan(start:f64,target:f64,cap:f64)->Dp54Times with Mut,Div {
 let direction=if target>start {1.0}else{-1.0}
 let remaining=dp_abs(target-start)
 let magnitude=dp_min(cap,remaining)
 var endpoint=if magnitude==remaining {target}else{start+direction*magnitude}
 if dp_abs(endpoint-start)>cap {endpoint=dp_adjacent(endpoint,-direction)}
 var times=dp_times(start,endpoint)
 if times.status!=0 {return times}
 if endpoint!=target {
  let trailing=dp_times(endpoint,target)
  if trailing.status==2 {
   let split=start+0.5*(target-start)
   let first=dp_times(start,split);let second=dp_times(split,target)
   if first.status!=0||second.status!=0||dp_abs(first.h)>cap||dp_abs(second.h)>cap {times.status=2;return times}
   times=first
  }
 }
 times
}

fn dp_error(state:&[f64;64],n:i64,h:f64,w:&Dp54Work64,opts:ODEOptions)->Dp54Value with Mut,Div,Panic {
 var i:i64=0
  var error:f64=0.0
  i=0
  while i<n {
   let defect=h*((71.0/57600.0)*w.k1[i as usize]-(71.0/16695.0)*w.k3[i as usize]+
    (71.0/1920.0)*w.k4[i as usize]-(17253.0/339200.0)*w.k5[i as usize]+
    (22.0/525.0)*w.k6[i as usize]-(1.0/40.0)*w.k7[i as usize])
   let denominator=opts.atol+opts.rtol*dp_max(dp_abs(state[i as usize]),dp_abs(w.trial[i as usize]))
   if !is_finite(defect) || !is_finite(denominator) || denominator<=0.0 {return dp_unavailable(4)}
   let normalized=dp_abs(defect)/denominator
   if !is_finite(normalized) {return dp_unavailable(4)}
   if normalized>error {error=normalized}
   i=i+1
  }
 Dp54Value{status:0,value:error}
}
fn dp_control(error:f64,opts:ODEOptions)->Dp54Value with Mut,Div,Panic {
 var factor=opts.max_growth
 if error>0.0 {let logged=ln_checked(error);if logged.status!=0 {return dp_unavailable(4)};factor=opts.safety*exp(-0.2*logged.value)}
 if !is_finite(factor)||factor<=0.0 {return dp_unavailable(4)}
 Dp54Value{status:0,value:dp_max(opts.min_shrink,dp_min(opts.max_growth,factor))}
}
fn dp_dense_preflight(state:&[f64;64],n:i64,h:f64,w:&!Dp54Work64)->i64 with Mut,Div,Panic {
 var i:i64=0
 while i<n {
  let r2=w.trial[i]-state[i]
  let r3=h*w.k1[i]-r2
  let r4=r2-h*w.k7[i]-r3
  let r5=h*((-12715105075.0/11282082432.0)*w.k1[i]+(87487479700.0/32700410799.0)*w.k3[i]+(-10690763975.0/1880347072.0)*w.k4[i]+(701980252875.0/199316789632.0)*w.k5[i]+(-1453857185.0/822651844.0)*w.k6[i]+(69997945.0/29380423.0)*w.k7[i])
  if !is_finite(r2)||!is_finite(r3)||!is_finite(r4)||!is_finite(r5) {return 4}
  w.k2[i]=r5;w.k3[i]=r2;w.k4[i]=r3;w.k5[i]=r4;i=i+1
 }
 0
}
fn dp_dense_commit(d:&!Dp54Dense64,state:&[f64;64],n:i64,times:Dp54Times,w:&Dp54Work64) with Mut,Panic {
 var i:i64=0
 while i<n {d.r1[i]=state[i];d.r2[i]=w.k3[i];d.r3[i]=w.k4[i];d.r4[i]=w.k5[i];d.r5[i]=w.k2[i];d.endpoint[i]=w.trial[i];d.slope0[i]=w.k1[i];d.slope1[i]=w.k7[i];i=i+1}
 d.n=n;d.t0=times.start;d.t1=times.end;d.status=0
}

fn dp_dense_query(d:&Dp54Dense64,t:f64,i:i64)->i64 {
 if d.status!=0 {return d.status}
 if i<0||i>=d.n||!is_finite(t) {return 3}
 if d.t0<d.t1 {if t<d.t0||t>d.t1 {return 3}}else{if t>d.t0||t<d.t1 {return 3}}
 0
}
pub fn dp54_dense_value(d:&Dp54Dense64,t:f64,i:i64)->Dp54Value with Div,Panic {
 let code=dp_dense_query(d,t,i)
 if code!=0 {return dp_unavailable(code)}
 if t==d.t0 {return Dp54Value{status:0,value:d.r1[i]}}
 if t==d.t1 {return Dp54Value{status:0,value:d.endpoint[i]}}
 let theta=(t-d.t0)/(d.t1-d.t0)
 let value=d.r1[i]+theta*(d.r2[i]+(1.0-theta)*(d.r3[i]+theta*(d.r4[i]+(1.0-theta)*d.r5[i])))
 if !is_finite(value) {return dp_unavailable(4)}

}
pub fn dp54_dense_derivative(d:&Dp54Dense64,t:f64,i:i64)->Dp54Value with Div,Panic {
 let code=dp_dense_query(d,t,i)
 if code!=0 {return dp_unavailable(code)}
 if t==d.t0 {return Dp54Value{status:0,value:d.slope0[i]}}
 if t==d.t1 {return Dp54Value{status:0,value:d.slope1[i]}}
 let h=d.t1-d.t0;let theta=(t-d.t0)/h;let u=1.0-theta
 let a=d.r4[i]+u*d.r5[i];let b=d.r3[i]+theta*a;let c=d.r2[i]+u*b
 let value=(c+theta*(-b+u*(a-theta*d.r5[i])))/h
 if !is_finite(value) {return dp_unavailable(4)}
 Dp54Value{status:0,value:value}
}
fn dp_limits_ok(l:Dp54Limits)->bool {l.max_rhs>=0&&l.max_rhs<=7000000&&l.max_event>=0&&l.max_event<=10000000&&l.max_root_iterations>=0&&l.max_root_iterations<=1000000}
fn dp_run_plain(rhs:fn(f64,&[f64;64],&![f64;64])->() with Mut,Div,Panic,y0:&[f64;64],n:i64,t0:f64,t_end:f64,opts:ODEOptions,limits:Dp54Limits,dense:&!Dp54Dense64,dense_on:bool)->Ode64Result with Mut,Div,Panic {
 dense.status=3;dense.n=0;dense.t0=0.0;dense.t1=0.0
 var out=dp_empty(n)
 if n<1||n>64||!is_finite(t0)||!is_finite(t_end)||!dp_options(opts)||!dp_limits_ok(limits) {return out}
 if !dp_finite(y0,n) {return out}
 out.t_final=t0
 var i:i64=0;while i<n {out.state[i]=y0[i];i=i+1}
 if t0==t_end {out.status=0;return out}
 let span=t_end-t0
 if !is_finite(span) {out.status=4;return out}
 var step=opts.dt_init;if step==0.0 {step=dp_abs(span)*0.01};step=dp_max(opts.dt_min,dp_min(opts.dt_max,step))
 var w=dp_work();var have_k1=false
 while out.t_final!=t_end&&out.attempts<opts.max_steps {
  let times=dp_plan(out.t_final,t_end,step)
  if times.status!=0 {out.status=times.status;return out}
  out.attempts=out.attempts+1
  let code=dp_trial(rhs,&out.state,n,times,&!w,&!out.nfeval,limits.max_rhs,have_k1)
  if code!=0 {out.status=code;return out}
  have_k1=true
  let err=dp_error(&out.state,n,times.h,&w,opts)
  if err.status!=0 {out.status=err.status;return out}
  let control=dp_control(err.value,opts)
  if control.status!=0 {out.status=control.status;return out}
  var factor=control.value
  if err.value<=1.0 {
   if dense_on {
    let ready=dp_dense_preflight(&out.state,n,times.h,&!w)
    if ready!=0 {out.status=ready;return out}
    dp_dense_commit(dense,&out.state,n,times,&w)
   }
   i=0;while i<n {out.state[i]=w.trial[i];w.k1[i]=w.k7[i];i=i+1}
   out.t_final=times.end;out.steps=out.steps+1
  }else{
   out.nreject=out.nreject+1;factor=dp_min(0.9,factor)
   if dp_abs(times.h)<=opts.dt_min {out.status=2;return out}
  }
  let scaled=dp_abs(times.h)*factor
  if !is_finite(scaled) {step=opts.dt_max}else{step=dp_max(opts.dt_min,dp_min(opts.dt_max,scaled))}
  if err.value>1.0&&step>=dp_abs(times.h) {out.status=2;return out}
 }
 if out.t_final!=t_end {out.status=1;return out}
 out.status=0;out
}
