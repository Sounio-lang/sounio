// PRIVATE diagnostic bridge only; never a production hook or source permission.
// Caller must already have explicitly selected the Linux exact-center policy.
pub fn ocp_private_real_owned_controls(module: &! IrModule, fi: i64) -> bool with Mut, Div, Panic {
    if OCP_RESEARCH_EXACT_INTEGRAL_CENTER != 1 { return false }
    if fi < 0 || fi >= (*module).fn_count { return false }
    let count = ir_fn_get(&(*module), fi).instr_count
    let regs = ir_fn_get(&(*module), fi).reg_count
    if count <= 0 || count > 128 { return false }
    let region = ir_fn_get(&(*module), fi).region
    let frame = ocp_center_frame(&(*module), fi, count, regs)
    if !frame.ok || !frame.owned { return false }
    let saved_count = OCP_RESEARCH_EXACT_CENTER_APPLICATIONS
    let saved_last = OCP_RESEARCH_EXACT_CENTER_LAST_FUNCTION
    var before1: [i64; 128] = [-1; 128]
    var before2: [i64; 128] = [-1; 128]
    var i: i64 = 0
    while i < count {
        let slot = ir_region_slot_r(region, i)
        before1[i as usize] = IR_A_SRC1[slot as usize]
        before2[i as usize] = IR_A_SRC2[slot as usize]
        i = i + 1
    }
    OCP_RESEARCH_EXACT_CENTER_APPLICATIONS = 0
    ocp_mfi_integral_center_pass(module, fi)
    var first: i64 = -1
    var second: i64 = -1
    var changed: i64 = 0
    i = 0
    while i < count {
        let slot = ir_region_slot_r(region, i)
        if IR_A_SRC1[slot as usize] != before1[i as usize]
            || IR_A_SRC2[slot as usize] != before2[i as usize] {
            changed = changed + 1
            if first < 0 { first = i } else { second = i }
        }
        i = i + 1
    }
    var ok = OCP_RESEARCH_EXACT_CENTER_APPLICATIONS == 1 && changed == 2
    if first >= 0 && second > first {
        let p = ir_region_slot_r(region, first)
        let q = ir_region_slot_r(region, second)
        if before1[second as usize] != IR_A_DST[p as usize]
            || IR_A_SRC1[p as usize] != before2[first as usize]
            || IR_A_SRC2[p as usize] != before2[second as usize]
            || IR_A_SRC1[q as usize] != before1[first as usize]
            || IR_A_SRC2[q as usize] != IR_A_DST[p as usize] { ok = false }
    }
    // Restore only actually changed source cells, even on failed baseline.
    i = 0
    while i < count {
        let slot = ir_region_slot_w(region, i)
        if IR_A_SRC1[slot as usize] != before1[i as usize] { IR_A_SRC1[slot as usize] = before1[i as usize] }
        if IR_A_SRC2[slot as usize] != before2[i as usize] { IR_A_SRC2[slot as usize] = before2[i as usize] }
        i = i + 1
    }
    OCP_RESEARCH_EXACT_CENTER_APPLICATIONS = saved_count
    OCP_RESEARCH_EXACT_CENTER_LAST_FUNCTION = saved_last
    if !ok || first < 0 || second <= first { return false }
    let inner = ir_region_slot_r(region, first)
    let outer = ir_region_slot_r(region, second)
    if (IR_A_OP[inner as usize] as IrOpcode) != IrOpcode::IrBinOp
        || (IR_A_OP[outer as usize] as IrOpcode) != IrOpcode::IrBinOp { return false }
    let entry = ir_region_slot_r(region, 0)
    let exit = ir_region_slot_r(region, count - 3)
    let target = IR_A_FN_ID[entry as usize]
    let entry_flag = IR_A_IMM_FLAGS[entry as usize]
    let entry_argc = IR_A_ARG_COUNT[entry as usize]
    let exit_opcode = IR_A_OP[exit as usize]
    let callee = ir_fn_get(&(*module), target)
    var alias_index: i64 = -1
    i = 0
    while i < (*module).fn_count {
        if i != fi && i != target && i != IR_A_FN_ID[exit as usize]
            && ir_fn_get(&(*module), i).instr_count == 0 { alias_index = i }
        i = i + 1
    }
    if alias_index < 0 { return false }
    let alias_before = ir_fn_get(&(*module), alias_index)
    var mode: i64 = 1
    while mode <= 6 {
        if mode == 1 { IR_A_IMM_FLAGS[entry as usize] = 0 }
        if mode == 2 { IR_A_ARG_COUNT[entry as usize] = 1 }
        if mode == 3 {
            var body_present = callee
            body_present.instr_count = 1
            ir_fn_set(module, target, body_present)
        }
        if mode == 4 { IR_A_OP[exit as usize] = IrOpcode::IrNop as i64 }
        if mode == 5 {
            var alias = alias_before
            alias.region = region
            ir_fn_set(module, alias_index, alias)
        }
        if mode == 6 { IR_A_SRC1[inner as usize] = IR_A_DST[exit as usize] }
        let expected1 = IR_A_SRC1[inner as usize]
        let expected2 = IR_A_SRC2[inner as usize]
        let expected3 = IR_A_SRC1[outer as usize]
        let expected4 = IR_A_SRC2[outer as usize]
        OCP_RESEARCH_EXACT_CENTER_APPLICATIONS = 0
        OCP_RESEARCH_EXACT_CENTER_LAST_FUNCTION = -1
        ocp_mfi_integral_center_pass(module, fi)
        if OCP_RESEARCH_EXACT_CENTER_APPLICATIONS != 0
            || IR_A_SRC1[inner as usize] != expected1 || IR_A_SRC2[inner as usize] != expected2
            || IR_A_SRC1[outer as usize] != expected3 || IR_A_SRC2[outer as usize] != expected4 { ok = false }
        IR_A_IMM_FLAGS[entry as usize] = entry_flag
        IR_A_ARG_COUNT[entry as usize] = entry_argc
        IR_A_OP[exit as usize] = exit_opcode
        ir_fn_set(module, target, callee)
        ir_fn_set(module, alias_index, alias_before)
        IR_A_SRC1[inner as usize] = before1[first as usize]
        IR_A_SRC2[inner as usize] = before2[first as usize]
        IR_A_SRC1[outer as usize] = before1[second as usize]
        IR_A_SRC2[outer as usize] = before2[second as usize]
        OCP_RESEARCH_EXACT_CENTER_APPLICATIONS = saved_count
        OCP_RESEARCH_EXACT_CENTER_LAST_FUNCTION = saved_last
        mode = mode + 1
    }
    // No alias remains and every SRC cell is restored, regardless of refusal result.
    i = 0
    while i < count {
        let slot = ir_region_slot_r(region, i)
        if IR_A_SRC1[slot as usize] != before1[i as usize]
            || IR_A_SRC2[slot as usize] != before2[i as usize] { ok = false }
        i = i + 1
    }
    ok && IR_A_IMM_FLAGS[entry as usize] == entry_flag
        && IR_A_ARG_COUNT[entry as usize] == entry_argc && IR_A_OP[exit as usize] == exit_opcode
        && ir_fn_get(&(*module), target).instr_count == callee.instr_count
        && ir_fn_get(&(*module), alias_index).region.slot == alias_before.region.slot
        && OCP_RESEARCH_EXACT_CENTER_APPLICATIONS == saved_count
        && OCP_RESEARCH_EXACT_CENTER_LAST_FUNCTION == saved_last
}
