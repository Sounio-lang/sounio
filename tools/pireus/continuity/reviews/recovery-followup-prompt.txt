// Bounded recovery authority: stop worker recreation only, never grant GPU access.
// Input: schema=1, fifteen boolean facts and one predecessor Slurm observation mask, two MemAvailable MiB.
// No grant, lease transition, barrier change, threshold override or resume action.
fn main() -> i32 with IO, Mut, Panic, Div {
    if arg_count() != 1 { return 2 }
    let path = get_arg(0)
    let size = file_size(path)
    if size <= 0 || size > 256 { return 2 }
    let raw = read_file(path)
    if str_len(raw) != size { return 2 }
    var values: [i64; 19] = [0; 19]
    var p: i64 = 0
    var i: i64 = 0
    while i < 19 {
        var digits: i64 = 0
        var value: i64 = 0
        while p < size {
            let c = str_char_at(raw, p)
            if c < 48 || c > 57 { break }
            if digits >= 9 { return 2 }
            value = value * 10 + c - 48
            digits = digits + 1
            p = p + 1
        }
        if digits == 0 || p >= size { return 2 }
        if str_char_at(raw, p) != (if i == 18 { 10 } else { 32 }) { return 2 }
        values[i as usize] = value
        p = p + 1
        i = i + 1
    }
    if p != size || values[0] != 1 { return 2 }
    var allowed = true
    i = 1
    while i <= 16 {
        if i == 15 {
            if values[15] > 511 { return 2 }
            if (values[15] & 30) != 30 { allowed = false }
        } else {
            if values[i as usize] > 1 { return 2 }
            if values[i as usize] != 1 { allowed = false }
        }
        i = i + 1
    }
    if values[17] < 32768 || values[18] < 32768 { allowed = false }
    print("{\"schema\":1,\"authority\":\"Sounio\",\"action\":\"DETACH_FENCED_WORKERS\",\"decision\":\"")
    if allowed { print("ALLOW") } else { print("REFUSE") }
    println("\",\"gpu_grant\":false,\"resume\":false,\"minimum_free_memory_mb\":32768,\"claim_ready\":false}")
    if allowed { 0 } else { 1 }
}
