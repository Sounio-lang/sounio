// A compound XOR recipe may replace only
// one straight-line aggregate store. Reject extra effects and pointer remaps.
pub fn gpu_xor_compound_shape_valid(kernel: GpuKernelIr) -> bool with Mut, Panic {
    var has_xor = false
    var scan: i64 = 0
    while scan < kernel.op_count {
        if kernel.ops[scan as usize].opcode == GpuOpcode::GpuXorConvolutionF64 { has_xor = true }
        scan = scan + 1
    }
    if !has_xor { return true }
    if kernel.param_count != 3 { return false }
    var params: [i64; 3] = [0 - 1; 3]
    var count: i64 = 0
    var labels: i64 = 0
    var seen: bool = false
    var i: i64 = 0
    while i < kernel.op_count {
        let op = kernel.ops[i as usize]
        if op.opcode == GpuOpcode::GpuLoadParam {
            if seen || op.param_idx < 0 || op.param_idx >= 3 { return false }
            params[op.param_idx as usize] = op.dst_reg
        } else if op.opcode == GpuOpcode::GpuXorConvolutionF64 {
            count = count + 1
            if count != 1 || params[0] < 0 || params[1] < 0 || params[2] < 0 { return false }
            if op.lhs_reg != params[0] || op.rhs_reg != params[1] || op.addr_reg != params[2] { return false }
            seen = true
        } else if op.opcode == GpuOpcode::GpuRet {
            if !seen || i != kernel.op_count - 1 { return false }
        } else if op.opcode == GpuOpcode::GpuBra {
            labels = labels + 1
            if seen || op.src_reg != 1 || labels > 1 { return false }
        } else {
            if seen { return false }
            // Dead aggregate-load scaffolding before the terminal store.
            let pure = op.opcode == GpuOpcode::GpuLoadGlobal
                || op.opcode == GpuOpcode::GpuAddImm
                || op.opcode == GpuOpcode::GpuMulImm
                || (op.opcode == GpuOpcode::GpuAdd && op.ty == GpuType::GpuU64)
            if !pure { return false }
        }
        i = i + 1
    }
    count == 1
}


FILE self-hosted/gpu/lower_to_ptx.sio
pub fn gpu_lower_to_ptx_for_target(kernel: GpuKernelIr, dgx_sm121: bool) -> PtxBuf with IO, Mut, Panic, Div, Alloc {
    var buf = ptx_buf_new()
    if !gpu_xor_compound_shape_valid(kernel) {
        return gpu_refuse_unlowered(buf, "PIREUS_PTX_COMPOUND_SHAPE_REFUSED")
    }

    // 1. Emit header (.version, .target, .address_size). A wmma m16n16k16 tile needs sm_70+
    // and a matching .version; the default header targets sm_50 (no tensor cores), so bump it.
    if dgx_sm121 {
        buf = ptx_emit_dgx_sm121_header(buf)
    } else if gpu_has_wmma_tile(kernel) {
        buf = ptx_push_str(buf, ".version 8.0\n.target sm_80\n.address_size 64\n\n")
    } else {
        buf = ptx_emit_header(buf)
    }

FILE self-hosted/gpu/lower_to_ptx.sio
fn gpu_refuse_unlowered(buf: PtxBuf, opcode_name: string) -> PtxBuf with IO, Mut, Panic, Div, Alloc {
    print("error: gpu codegen: no PTX lowering rule for GpuOpcode::")
    print(opcode_name)
    print(" - refusing to emit PTX (see issue #2221)\n")
    panic("gpu codegen: unlowered GpuOpcode reached PTX lowering")
    buf
}
